cmake_minimum_required(VERSION 3.15...3.31)
cmake_policy(SET CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(recorddy-rife-stream LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenMP REQUIRED)
find_package(Vulkan REQUIRED)
# Se conserva la selección de operadores del ejecutable RIFE de referencia.
file(STRINGS "${RIFE_SRC}/CMakeLists.txt" options REGEX "^[ ]+option\\((NCNN_|WITH_LAYER_)")
foreach(line IN LISTS options)
  cmake_language(EVAL CODE "${line}")
endforeach()
add_subdirectory("${NCNN_SRC}" ncnn)
file(GLOB shaders "${RIFE_SRC}/*.comp")
foreach(shader IN LISTS shaders)
  get_filename_component(name "${shader}" NAME)
  set(header "${CMAKE_CURRENT_BINARY_DIR}/${name}.hex.h")
  add_custom_command(OUTPUT "${header}" COMMAND "${CMAKE_COMMAND}" "-DSHADER_SRC=${shader}" "-DSHADER_COMP_HEADER=${header}" -P "${RIFE_SRC}/generate_shader_comp_header.cmake" DEPENDS "${shader}" VERBATIM)
  list(APPEND generated "${header}")
endforeach()
add_executable(recorddy-rife-stream main.cpp "${RIFE_SRC}/rife.cpp" "${RIFE_SRC}/warp.cpp" ${generated})
target_include_directories(recorddy-rife-stream PRIVATE "${RIFE_SRC}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(recorddy-rife-stream PRIVATE ncnn OpenMP::OpenMP_CXX Vulkan::Vulkan)
